# Theme Customization + Store Readiness Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let users customize the dark and light themes (background/text colors, contrast, saturation, keep-colors), reset to defaults, and make the extension Chrome-Web-Store-ready with tests, docs, packaging and CI.

**Architecture:** Pure-function core (`lib/`: color, defaults, settings normalization, theme resolution, composed-matrix filter builder, store merge) shared by content script, popup and options page as classic scripts on a `self.SL` namespace (Node-testable via `module.exports`). Content script injects one inline SVG with a forward and an inverse `feColorMatrix` and drives `html[data-sl-theme]` + CSS variables. Popup and options write field-level patches through `store.save(patch)`.

**Tech Stack:** Manifest V3, vanilla JS/CSS (no bundler), `node:test` for unit tests, `@playwright/test` for e2e (Chromium with the unpacked extension), ESLint + Prettier, GitHub Actions.

**Spec:** `docs/superpowers/specs/2026-08-24-theme-customization-and-store-readiness-design.md`

## Global Constraints

- Only permission: `storage`. Content script matches only `*://teams.scrumlaunch.com/*`. No `host_permissions`, no background worker, no remote code.
- No build step: `src/` must be loadable unpacked as-is.
- Library files: classic scripts attaching to `self.SL` in browsers and `module.exports` in Node. Load order: color → defaults → settings → theme-logic → filter → settings-store.
- Settings schema v2 exactly as in the spec; v1 `{ mode }` migrates transparently; write-back only from the top frame.
- Contrast/saturation range 50..150; hex colors strict `#rrggbb` lowercased; `keepColors = v == null ? default : Boolean(v)`.
- Product name "Dark Modern for ScrumLaunch Teams"; version `2.0.0`; `minimum_chrome_version` `111`; `default_locale` `en`.
- Every `store.save()` caller surfaces rejection as "Not saved — <message>".
- Tests live in the task that builds the code (TDD for pure code, e2e alongside each surface). Commit after every task with a conventional-commit message and the Co-Authored-By trailer.

```js
(function (root) {
  const { normalizeSettings } = root.SL ? root.SL.settings : require('./settings.js');
  const { SETTINGS_VERSION } = root.SL ? root.SL.defaults : require('./defaults.js');
  function merge(base, patch) {
    const out = { ...base, ...patch, themes: { ...base.themes } };
    for (const k of ['dark', 'light'])
      if (patch.themes && patch.themes[k])
        out.themes[k] = { ...base.themes[k], ...patch.themes[k] };
    return out;
  }
  function read() {
    return new Promise((res) => chrome.storage.sync.get(['settings', 'mode'], res));
  }
  function write(settings, removeKeys) {
    return new Promise((res, rej) =>
      chrome.storage.sync.set({ settings }, () => {
        if (chrome.runtime.lastError) return rej(new Error(chrome.runtime.lastError.message));
        if (removeKeys) chrome.storage.sync.remove(removeKeys, () => res(settings));
        else res(settings);
      }),
    );
  }
  async function load() {
    const raw = await read();
    const isV2 = !!raw.settings && raw.settings.v === SETTINGS_VERSION;
    const settings = normalizeSettings(isV2 ? raw.settings : raw.settings || { mode: raw.mode });
    if (!isV2 && typeof window !== 'undefined' && window === window.top) {
      try {
        await write(settings, ['mode']);
      } catch (e) {
        console.warn('[SL] migration write failed', e);
      }
    }
    return settings;
  }
  async function save(patch) {
    const fresh = await load();
    return write(normalizeSettings(merge(fresh, patch)));
  }
  function onChange(cb) {
    chrome.storage.onChanged.addListener((ch, area) => {
      if (area === 'sync' && ch.settings) cb(normalizeSettings(ch.settings.newValue));
    });
  }
  const api = { merge, load, save, onChange };
  if (typeof module !== 'undefined' && module.exports) module.exports = api;
  else {
    root.SL = root.SL || {};
    root.SL.store = api;
  }
})(typeof self !== 'undefined' ? self : this);
```

```js
(function () {
  const { resolveTheme } = SL.theme,
    { buildFilter, FILTER_ID } = SL.filter,
    store = SL.store;
  const SVG_NS = 'http://www.w3.org/2000/svg';
  const media = matchMedia('(prefers-color-scheme: dark)');
  let settings = null;
  function makeFilter(svg, id) {
    const f = document.createElementNS(SVG_NS, 'filter');
    f.id = id;
    f.setAttribute('color-interpolation-filters', 'sRGB');
    const m = document.createElementNS(SVG_NS, 'feColorMatrix');
    m.setAttribute('type', 'matrix');
    f.appendChild(m);
    svg.appendChild(f);
    return m;
  }
  function ensureSvg() {
    let svg = document.getElementById('sl-theme-svg');
    if (svg)
      return {
        fwd: svg.querySelector(`#${FILTER_ID} feColorMatrix`),
        inv: svg.querySelector(`#${FILTER_ID}-inverse feColorMatrix`),
      };
    svg = document.createElementNS(SVG_NS, 'svg');
    svg.id = 'sl-theme-svg';
    svg.setAttribute('aria-hidden', 'true');
    svg.setAttribute(
      'style',
      'position:absolute;width:0;height:0;overflow:hidden;pointer-events:none',
    );
    const fwd = makeFilter(svg, FILTER_ID),
      inv = makeFilter(svg, `${FILTER_ID}-inverse`);
    document.documentElement.appendChild(svg);
    return { fwd, inv };
  }
  function apply() {
    if (!settings) return;
    const theme = settings.themes[resolveTheme(settings.mode, media.matches)];
    const { css, inverseCss, matrix, inverseMatrix, inverted } = buildFilter(theme);
    const root = document.documentElement;
    if (css === 'none') {
      root.removeAttribute('data-sl-theme');
      for (const p of ['--sl-filter', '--sl-filter-inverse', '--sl-bg'])
        root.style.removeProperty(p);
      return;
    }
    const { fwd, inv } = ensureSvg();
    fwd.setAttribute('values', matrix.join(' '));
    inv.setAttribute('values', inverseMatrix.join(' '));
    root.style.setProperty('--sl-filter', css);
    root.style.setProperty('--sl-filter-inverse', inverseCss);
    root.style.setProperty('--sl-bg', theme.background);
    root.setAttribute('data-sl-theme', inverted ? 'dark' : 'light');
  }
  store.load().then((s) => {
    settings = s;
    apply();
  });
  store.onChange((s) => {
    settings = s;
    apply();
  });
  media.addEventListener('change', apply);
})();
```
